## 1. What are today's architecture decisions?

A RAG backend has two separate paths:

- **Ingestion path:** Load documents, split text into chunks, create embeddings, and store chunks with metadata.
- **Retrieval path:** Accept a question, search stored chunks, and return relevant text with metadata.

Keeping these responsibilities separate makes the system easier to test and change. For this first version, use a small local text corpus, Sentence Transformers for embeddings, and ChromaDB for vector storage. Test each path before adding generation or an API.

## 2. Today's backend flow

```text
data/documents.txt
       |
       v
  Load and chunk text
       |
       v
 Sentence Transformers
       |
       v
    ChromaDB
       ^
       |
 User question -> Query embedding -> Similarity search
                                      |
                                      v
                            Relevant chunks + metadata
```

## 3. Implement document ingestion

Create or replace `src/ingestion.py`:

```python
from pathlib import Path

import chromadb
from sentence_transformers import SentenceTransformer

DATA_PATH = Path("data/documents.txt")
CHROMA_PATH = "chroma_data"
COLLECTION_NAME = "knowledge_base"
EMBEDDING_MODEL = "sentence-transformers/all-MiniLM-L6-v2"


def load_text(path: Path = DATA_PATH) -> str:
    if not path.exists():
        raise FileNotFoundError(
            f"Document file not found: {path}. Create it and add some text first."
        )
    text = path.read_text(encoding="utf-8").strip()
    if not text:
        raise ValueError(f"Document file is empty: {path}")
    return text


def chunk_text(text: str, chunk_size: int = 700, overlap: int = 100) -> list[str]:
    if chunk_size <= 0:
        raise ValueError("chunk_size must be greater than zero")
    if overlap < 0 or overlap >= chunk_size:
        raise ValueError("overlap must be >= 0 and smaller than chunk_size")

    chunks = []
    start = 0
    while start < len(text):
        chunk = text[start:start + chunk_size].strip()
        if chunk:
            chunks.append(chunk)
        start += chunk_size - overlap
    return chunks


def ingest_documents() -> int:
    chunks = chunk_text(load_text())
    model = SentenceTransformer(EMBEDDING_MODEL)
    embeddings = model.encode(chunks, convert_to_numpy=True).tolist()

    client = chromadb.PersistentClient(path=CHROMA_PATH)
    collection = client.get_or_create_collection(name=COLLECTION_NAME)

    ids = [f"documents_chunk_{i}" for i in range(len(chunks))]
    metadatas = [
        {"source": DATA_PATH.name, "chunk_index": i}
        for i in range(len(chunks))
    ]

    collection.upsert(
        ids=ids,
        documents=chunks,
        embeddings=embeddings,
        metadatas=metadatas,
    )
    return len(chunks)


if __name__ == "__main__":
    count = ingest_documents()
    print(f"Ingested {count} chunks into ChromaDB.")
```

## 4. Implement retrieval

Create or replace `src/retrieval.py`:

```python
import chromadb
from sentence_transformers import SentenceTransformer

CHROMA_PATH = "chroma_data"
COLLECTION_NAME = "knowledge_base"
EMBEDDING_MODEL = "sentence-transformers/all-MiniLM-L6-v2"


def retrieve(query: str, k: int = 3) -> dict:
    if not query or not query.strip():
        raise ValueError("query must not be empty")
    if k < 1:
        raise ValueError("k must be at least 1")

    client = chromadb.PersistentClient(path=CHROMA_PATH)
    collection = client.get_or_create_collection(name=COLLECTION_NAME)

    if collection.count() == 0:
        raise RuntimeError(
            "The knowledge base is empty. Run ingestion first: "
            "python -m src.ingestion"
        )

    model = SentenceTransformer(EMBEDDING_MODEL)
    query_embedding = model.encode([query], convert_to_numpy=True).tolist()

    return collection.query(
        query_embeddings=query_embedding,
        n_results=min(k, collection.count()),
        include=["documents", "metadatas", "distances"],
    )


if __name__ == "__main__":
    question = input("Ask a question about the documents: ").strip()
    results = retrieve(question, k=3)

    for rank, document in enumerate(results["documents"][0], start=1):
        metadata = results["metadatas"][0][rank - 1]
        distance = results["distances"][0][rank - 1]
        print(
            f"\nResult {rank} | source={metadata.get('source')} "
            f"| distance={distance:.4f}"
        )
        print(document)
```

This first version loads the embedding model every time a function is called. That's acceptable for a small learning prototype; later, load it once and reuse it.

## Key takeaways

- Ingestion prepares and stores document chunks; retrieval finds relevant chunks for a query.
- Store source metadata so retrieved evidence can be traced.
- Test retrieval with questions whose answers exist in the sample documents.
- Inspect returned text; a similarity score alone does not prove relevance.
- Keep the first implementation small and improve it through evaluation.
